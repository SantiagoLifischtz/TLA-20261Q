program -> definitions

definitions -> definition definitions | %empty

definition -> config_sentence | pattern_definition | track | play

play -> OPEN_BRACE track_ids SEMICOLON CLOSE_BRACE | ALL SEMICOLON

track_ids -> ID | ID COMMA track_ids

config_sentence -> config_sentence_content SEMICOLON

config_sentence_content -> key | tempo

tempo -> TEMPO float

key -> KEY pitch mode

mode -> IONIAN | MAJOR | DORIAN | PHRYGIAN | LYDIAN | MIXOLYDIAN | AEOLIAN | MINOR | LOCRIAN

pitch -> note_letter semitone_modifier

note_letter -> C | D | E | F | G | A | B

semitone_modifier -> SHARP | FLAT | NATURAL | %empty

track -> TRACK ID OPEN_BRACE instrument sentences CLOSE_BRACE

pattern_definition -> PATTERN ID block

instrument -> INSTRUMENT string SEMICOLON

sentences -> sentence sentences | %empty

sentence -> pattern_sentence | tempo | key

pattern -> inline_pattern wait_after_inline | block wait_after_block | repeat | step

wait_after_inline -> duration SEMICOLON | SEMICOLON

wait_after_block -> duration SEMICOLON | %empty

inline_pattern -> note | rest | chord | strum | arpeggio | degree

note -> note_id duration

note_id -> pitch integer

duration -> expression

rest -> REST duration

repeat -> REPEAT integer pattern

step -> STEP duration pattern

block -> OPEN_BRACE sentences CLOSE_BRACE

chord -> pattern_chord | note_chord

pattern_chord -> OPEN_SQUARE comma_separated_patterns CLOSE_SQUARE

note_chord -> OPEN_SQUARE comma_separated_notes CLOSE_SQUARE duration

comma_separated_patterns -> pattern | pattern COMMA comma_separated_patterns

comma_separated_notes -> note_id | note_id COMMA comma_separated_notes

strum -> STRUM OPEN_PARENTHESIS comma_separated_notes COMMA duration COMMA duration CLOSE_PARENTHESIS

arpeggio -> ARPEGGIO OPEN_PARENTHESIS comma_separated_notes COMMA duration CLOSE_PARENTHESIS

degree -> DEGREE OPEN_PARENTHESIS integer COMMA integer CLOSE_PARENTHESIS semitone_modifier duration



float -> ...

integer -> ...

string -> ...

expression -> sum | product | subtraction | division | grouped_expression | dotted | number

number -> float | integer | standard_time

sum -> expression PLUS expression

product -> expression ASTERISK expression

subtraction -> expression MINUS expression

division -> expression SLASH expression

grouped_expression -> OPEN_PARENTHESIS expression CLOSE_PARENTHESIS

dotted -> expression DOT

standard_time -> WHOLE | W | HALF | H | QUARTER | Q | EIGHTH | E | SIXTEENTH | S | THIRTYSECOND | THS | SIXTYFOURTH | SXF
